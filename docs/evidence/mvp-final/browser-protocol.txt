const {chromium}=require('/private/tmp/live-jvg-mvp-browser/node_modules/playwright');
const assert=require('node:assert/strict');const fs=require('node:fs');
(async()=>{
 const browser=await chromium.launch({headless:true});const page=await browser.newPage({viewport:{width:1440,height:1000}});
 const errors=[];page.on('pageerror',e=>errors.push(e.message));
 await page.goto('http://localhost:18080',{waitUntil:'networkidle'});
 await page.waitForFunction(()=>document.getElementById('connection').textContent==='Conectado al servidor');
 const initial=await page.evaluate(async()=>{let a=await(await fetch('/api/matches')).json();return await(await fetch('/api/matches/'+a[0].match_id)).json()});
 const h=initial.home_score,a=initial.away_score;const steps=[];
 async function step(button,type,home,away,status){
  const responsePromise=page.waitForResponse(r=>r.request().method()==='POST'&&r.url().includes('/events'));
  await page.locator(button).click();const response=await responsePromise;assert.equal(response.status(),202,await response.text());const published=await response.json();
  await page.waitForFunction(({home,away,status})=>Number(document.getElementById('home-score').textContent)===home&&Number(document.getElementById('away-score').textContent)===away&&document.getElementById('var-state').classList.contains('review')===(status==='IN_PROGRESS')&&document.getElementById('feedback').textContent.includes('recibido'),{home,away,status},{timeout:20000});
  const state=await page.evaluate(async id=>await(await fetch('/api/matches/'+id)).json(),initial.match_id);
  assert(state.events.some(e=>e.event_id===published.event_id));assert.equal(state.var_status,status);
  steps.push({type,published,home_score:state.home_score,away_score:state.away_score,var_status:state.var_status});console.log('PASS UI',type,home,away,status,published.event_id);
 }
 await step('#goal-home','GOAL',h+1,a,'CLEAR');
 await step('#var-check','VAR_CHECK',h+1,a,'IN_PROGRESS');
 await step('#annul','GOAL_ANNULLED',h,a,'CLEAR');
 await step('#continue','MATCH_CONTINUES',h,a,'CLEAR');
 await step('#goal-away','GOAL',h,a+1,'CLEAR');
 assert.equal(steps[1].published.reference_event_id,steps[0].published.event_id);assert.equal(steps[2].published.reference_event_id,steps[0].published.event_id);
 await page.reload({waitUntil:'networkidle'});await page.waitForFunction(({h,a})=>Number(document.getElementById('home-score').textContent)===h&&Number(document.getElementById('away-score').textContent)===a,{h,a:a+1});
 await page.screenshot({path:'/private/tmp/live-jvg-mvp/desktop.png',fullPage:true});
 await page.setViewportSize({width:390,height:844});await page.screenshot({path:'/private/tmp/live-jvg-mvp/mobile.png',fullPage:true});
 assert(await page.evaluate(()=>document.documentElement.scrollWidth<=window.innerWidth+1),'Mobile overflow');assert.deepEqual(errors,[]);
 fs.writeFileSync('/private/tmp/live-jvg-mvp/browser-evidence.json',JSON.stringify({initial,steps,page_errors:errors,desktop:'desktop.png',mobile:'mobile.png'},null,2));
 console.log('PASS browser reload, desktop, mobile, no JavaScript errors');await browser.close();
})().catch(e=>{console.error(e);process.exit(1)});
