public class Robustness {
 static String run="phase7f-"+UUID.randomUUID(), exchange=run+"-updates";
 static final com.fasterxml.jackson.databind.ObjectMapper mapper=new MessagingConfiguration().messagingObjectMapper();
 static final List<ConfigurableApplicationContext> contexts=new ArrayList<>();
 static void check(boolean value,String name) { if(!value) throw new AssertionError(name); System.out.println("CHECK PASS "+name); }
 static void until(BooleanSupplier p,String label) throws Exception { long end=System.nanoTime()+TimeUnit.SECONDS.toNanos(20); while(!p.getAsBoolean() && System.nanoTime()<end) Thread.sleep(100);check(p.getAsBoolean(),label); }
 static ConfigurableApplicationContext start(String topic,String match,String group) throws Exception {
  var c=new SpringApplicationBuilder(MatchStateServiceApplication.class).web(WebApplicationType.NONE).run(
   "--spring.kafka.bootstrap-servers=localhost:9092","--match-state.kafka.topic="+topic,
   "--spring.kafka.consumer.group-id="+group,"--spring.kafka.consumer.auto-offset-reset=earliest",
   "--match-state.rabbit.exchange="+exchange,"--spring.rabbitmq.host=localhost",
   "--spring.rabbitmq.port="+System.getProperty("test.rabbit.port","5672"),"--spring.rabbitmq.username=guest","--spring.rabbitmq.password=guest",
   "--match-state.matches[0].match-id="+match,"--match-state.matches[0].home-team-id=home",
   "--match-state.matches[0].away-team-id=away","--match-state.kafka.retry-interval=200ms",
   "--match-state.rabbit.confirm-timeout=1s","--spring.jmx.enabled=false");
  contexts.add(c);
  until(()->c.getBean(KafkaListenerEndpointRegistry.class).getListenerContainers().stream().anyMatch(x->x.getAssignedPartitions()!=null && !x.getAssignedPartitions().isEmpty()),"assigned "+topic);return c;
 }
 static InMemoryMatchStateStore store(ConfigurableApplicationContext c){return c.getBean(InMemoryMatchStateStore.class);}
 static MatchEvent ev(String m,EventType t,String team,String ref){return MatchEvent.create(m,t,team,10,Instant.now(),ref);}
 static long committed(AdminClient a,String group,String topic)throws Exception {
  var r=a.listConsumerGroupOffsets(group).partitionsToOffsetAndMetadata().get(5,TimeUnit.SECONDS);
  var o=r.get(new org.apache.kafka.common.TopicPartition(topic,0));return o==null?-1:o.offset();
 }
 static void offset(AdminClient a,String g,String t,long n)throws Exception {
  until(()->{try{return committed(a,g,t)==n;}catch(Exception e){return false;}},"offset="+n+" "+t);
 }
 public static void main(String[] args)throws Exception {
  System.out.println("RUN "+run);
  var cf=new CachingConnectionFactory("localhost",5672);cf.setUsername("guest");cf.setPassword("guest");
  cf.setPublisherConfirmType(CachingConnectionFactory.ConfirmType.CORRELATED);cf.setPublisherReturns(true);
  var rabbit=new RabbitTemplate(cf);rabbit.setMandatory(true);var ra=new RabbitAdmin(cf);
  var ex=new TopicExchange(exchange,true,false);ra.declareExchange(ex);
  Map<String,Object> props=new HashMap<>();props.put("bootstrap.servers","localhost:9092");
  props.put("key.serializer","org.apache.kafka.common.serialization.StringSerializer");
  props.put("value.serializer","org.apache.kafka.common.serialization.StringSerializer");props.put("acks","all");props.put("enable.idempotence",true);
  var pf=new DefaultKafkaProducerFactory<String,String>(props);var kt=new KafkaTemplate<String,String>(pf);
  try(var admin=AdminClient.create(Map.of("bootstrap.servers","localhost:9092"))) {
   String topic=run+"-main",match=run+"-match",group=run+"-group",queue=run+"-queue";
   String bad=run+"-invalid",early=run+"-early";
   admin.createTopics(List.of(new NewTopic(topic,1,(short)1),new NewTopic(bad,1,(short)1),new NewTopic(early,1,(short)1))).all().get(10,TimeUnit.SECONDS);
   ra.declareQueue(new Queue(queue,true,false,false));
   var c=start(topic,match,group);var producer=new KafkaMatchEventPublisher(kt,mapper,topic);
   var goal=ev(match,EventType.GOAL,"home",null);
   producer.publish(goal).get(10,TimeUnit.SECONDS);
   until(()->store(c).pendingForMatch(match).size()==1,"unroutable retains pending");
   Thread.sleep(700);check(committed(admin,group,topic)<1,"unroutable blocks commit");
   check(store(c).findById(match).orElseThrow().score().home()==1,"unroutable applies goal once");
   ra.declareBinding(BindingBuilder.bind(new Queue(queue)).to(ex).with("score."+match));
   offset(admin,group,topic,1);check(store(c).pendingForMatch(match).isEmpty(),"binding recovery clears pending");
   producer.publish(goal).get(10,TimeUnit.SECONDS);offset(admin,group,topic,2);
   var annul=ev(match,EventType.GOAL_ANNULLED,null,goal.eventId());producer.publish(annul).get(10,TimeUnit.SECONDS);offset(admin,group,topic,3);
   producer.publish(annul).get(10,TimeUnit.SECONDS);offset(admin,group,topic,4);
   check(store(c).findById(match).orElseThrow().score().home()==0,"duplicate annulment no negative score");
   check(ra.getQueueInfo(queue).getMessageCount()==2,"duplicates yield two outputs for four records");
   var g2=ev(match,EventType.GOAL,"home",null);producer.publish(g2).get(10,TimeUnit.SECONDS);offset(admin,group,topic,5);
   check(store(c).findById(match).orElseThrow().score().home()==1,"pre-restart home score=1");
   c.close();var restarted=start(topic,match,group);
   check(store(restarted).findById(match).orElseThrow().score().home()==0,"OBSERVED state lost after restart");
   var away=ev(match,EventType.GOAL,"away",null);producer.publish(away).get(10,TimeUnit.SECONDS);offset(admin,group,topic,6);
   var score=store(restarted).findById(match).orElseThrow().score();
   check(score.home()==0 && score.away()==1,"OBSERVED post-restart score=0-1 instead of durable 1-1");
   String bg=run+"-invalid-group",eg=run+"-early-group";
   var bc=start(bad,match,bg);var ec=start(early,match,eg);
   kt.send(bad,match,"{invalid-json").get(10,TimeUnit.SECONDS);
   new KafkaMatchEventPublisher(kt,mapper,bad).publish(ev(match,EventType.GOAL,"home",null)).get(10,TimeUnit.SECONDS);
   var futureGoal=ev(match,EventType.GOAL,"home",null);
   var ep=new KafkaMatchEventPublisher(kt,mapper,early);
   ep.publish(ev(match,EventType.GOAL_ANNULLED,null,futureGoal.eventId())).get(10,TimeUnit.SECONDS);
   ep.publish(futureGoal).get(10,TimeUnit.SECONDS);
   Thread.sleep(2500);
   check(committed(admin,bg,bad)<1 && store(bc).findById(match).orElseThrow().processedEvents().isEmpty(),"invalid JSON blocks later goal");
   check(committed(admin,eg,early)<1 && store(ec).findById(match).orElseThrow().processedEvents().isEmpty(),"early annulment blocks later original goal");
   var rp=new RabbitScoreUpdatePublisher(rabbit,mapper,exchange,"score.{matchId}",Duration.ofSeconds(1));
   String rm=run+"-connection",rq=run+"-connection-queue";ra.declareQueue(new Queue(rq,true,false,false));ra.declareBinding(BindingBuilder.bind(new Queue(rq)).to(ex).with("score."+rm));
   var mem=new InMemoryMatchStateStore(List.of(MatchState.start(rm,"home","away")));
   var badcf=new CachingConnectionFactory("localhost",unusedPort());badcf.setConnectionTimeout(500);
   badcf.setPublisherConfirmType(CachingConnectionFactory.ConfirmType.CORRELATED);badcf.setPublisherReturns(true);
   var badrt=new RabbitTemplate(badcf);badrt.setMandatory(true);
   var badrp=new RabbitScoreUpdatePublisher(badrt,mapper,exchange,"score.{matchId}",Duration.ofSeconds(1));
   var rg=new com.live_jgv.matchstate.domain.model.MatchEvent("connection-goal",rm,com.live_jgv.matchstate.domain.model.EventType.GOAL,"home",1,Instant.now(),null);
   boolean failure=false;try{new ProcessAndPublishMatchEvent(new MatchStateService(mem),mem,badrp).process(rg);}catch(RuntimeException e){failure=true;System.out.println("EXPECTED "+e.getClass().getSimpleName());}
   check(failure && mem.pendingForMatch(rm).size()==1,"Rabbit connection refusal retains pending");
   new ProcessAndPublishMatchEvent(new MatchStateService(mem),mem,rp).process(rg);
   check(mem.findById(rm).orElseThrow().score().home()==1 && mem.pendingForMatch(rm).isEmpty() && ra.getQueueInfo(rq).getMessageCount()==1,"Rabbit connection restored one output no double goal");badcf.destroy();
   var offlineProps=new HashMap<>(props);offlineProps.put("bootstrap.servers","localhost:"+unusedPort());offlineProps.put("max.block.ms",1000);offlineProps.put("request.timeout.ms",1000);offlineProps.put("delivery.timeout.ms",3000);
   var offpf=new DefaultKafkaProducerFactory<String,String>(offlineProps);boolean kfail=false;
   try{new KafkaMatchEventPublisher(new KafkaTemplate<>(offpf),mapper,run+"-offline").publish(ev(rm,EventType.GOAL,"home",null)).get(8,TimeUnit.SECONDS);}catch(Exception e){kfail=true;System.out.println("EXPECTED Kafka "+e.getClass().getSimpleName());}finally{offpf.destroy();}
   check(kfail,"Kafka unreachable endpoint fails future");
   System.out.println("RESOURCES "+topic+" "+bad+" "+early+" "+queue+" "+rq+" "+exchange);
   System.out.println("ROBUSTNESS_COMPLETED");
  } finally {for(var c:contexts)if(c.isActive())c.close();pf.destroy();cf.destroy();}
 }
 static int unusedPort()throws Exception{try(var s=new ServerSocket(0)){return s.getLocalPort();}}
}
