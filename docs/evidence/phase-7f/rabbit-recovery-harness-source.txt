import java.net.*;
import java.io.*;
import java.util.*;
import java.util.concurrent.*;
import org.springframework.amqp.rabbit.core.*;
import org.springframework.amqp.rabbit.connection.CachingConnectionFactory;
import org.springframework.amqp.core.*;
import org.apache.kafka.clients.admin.*;
import org.springframework.kafka.core.*;
import com.live_jgv.live_feed_producer.infrastructure.kafka.KafkaMatchEventPublisher;
import com.live_jgv.live_feed_producer.domain.model.*;

public class RabbitRecovery {
 static class Proxy implements AutoCloseable {
  final ServerSocket listener=new ServerSocket(0);volatile boolean enabled=true,closed=false;
  final Set<Socket> sockets=ConcurrentHashMap.newKeySet();
  Proxy()throws Exception {Thread t=new Thread(()->{while(!closed)try{
   Socket client=listener.accept();if(!enabled){client.close();continue;}
   Socket broker=new Socket("localhost",5672);sockets.add(client);sockets.add(broker);
   pipe(client,broker);pipe(broker,client);
  }catch(IOException e){if(!closed)e.printStackTrace();}});t.setDaemon(true);t.start();}
  void pipe(Socket a,Socket b){Thread t=new Thread(()->{try{a.getInputStream().transferTo(b.getOutputStream());}catch(IOException e){}finally{try{a.close();b.close();}catch(IOException e){}sockets.remove(a);sockets.remove(b);}});t.setDaemon(true);t.start();}
  void cut(){enabled=false;for(Socket s:sockets)try{s.close();}catch(IOException e){}}
  public void close()throws Exception{closed=true;cut();listener.close();}
 }
 public static void main(String[] args)throws Exception{
  try(var proxy=new Proxy();var admin=AdminClient.create(Map.of("bootstrap.servers","localhost:9092"))){
   String topic=Robustness.run+"-network",match=Robustness.run+"-network-match",group=Robustness.run+"-network-group",queue=Robustness.run+"-network-queue";
   System.out.println("NETWORK_RUN "+Robustness.run+" proxyPort="+proxy.listener.getLocalPort());
   System.setProperty("test.rabbit.port",Integer.toString(proxy.listener.getLocalPort()));
   var cf=new CachingConnectionFactory("localhost",5672);var ra=new RabbitAdmin(cf);var ex=new TopicExchange(Robustness.exchange,true,false);
   ra.declareExchange(ex);ra.declareQueue(new org.springframework.amqp.core.Queue(queue,true,false,false));ra.declareBinding(BindingBuilder.bind(new org.springframework.amqp.core.Queue(queue)).to(ex).with("score."+match));
   var pf=new DefaultKafkaProducerFactory<String,String>(Map.of("bootstrap.servers","localhost:9092","key.serializer","org.apache.kafka.common.serialization.StringSerializer","value.serializer","org.apache.kafka.common.serialization.StringSerializer","acks","all"));
   try {
    admin.createTopics(List.of(new NewTopic(topic,1,(short)1))).all().get(10,TimeUnit.SECONDS);
    var c=Robustness.start(topic,match,group);var publisher=new KafkaMatchEventPublisher(new KafkaTemplate<>(pf),Robustness.mapper,topic);
    publisher.publish(Robustness.ev(match,EventType.GOAL,"home",null)).get(10,TimeUnit.SECONDS);Robustness.offset(admin,group,topic,1);
    proxy.cut();
    publisher.publish(Robustness.ev(match,EventType.GOAL,"away",null)).get(10,TimeUnit.SECONDS);
    Robustness.until(()->Robustness.store(c).pendingForMatch(match).size()==1,"AMQP cut retains pending");Thread.sleep(1800);
    Robustness.check(Robustness.committed(admin,group,topic)==1,"AMQP cut freezes offset=1");
    Robustness.check(ra.getQueueInfo(queue).getMessageCount()==1,"no output while AMQP cut");
    proxy.enabled=true;Robustness.offset(admin,group,topic,2);
    Robustness.check(Robustness.store(c).pendingForMatch(match).isEmpty(),"AMQP reconnect clears pending");
    Robustness.check(ra.getQueueInfo(queue).getMessageCount()==2,"AMQP reconnect exactly two outputs");
    var score=Robustness.store(c).findById(match).orElseThrow().score();Robustness.check(score.home()==1&&score.away()==1,"AMQP reconnect score 1-1 no double effect");
    System.out.println("NETWORK_RESOURCES "+topic+" "+group+" "+queue+" "+Robustness.exchange);
    System.out.println("NETWORK_RECOVERY_COMPLETED");
   }finally{for(var c:Robustness.contexts)if(c.isActive())c.close();pf.destroy();cf.destroy();}
  }
 }
}
